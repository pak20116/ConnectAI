 2	import { styled } from '@emotion/styled';
 3	
 4	interface StabilityData {
 5	  latencyMs: number;
 6	  visualConsistencyScore: number;
 7	}
 8	
 9	interface StabilityGaugeProps {
10	  data: StabilityData;
11	  title: string;
12	  unit: string;
13	  colorScheme: 'success' | 'warning' | 'danger';
14	  benchmark: number; // For latency, e.g., 100ms

16	}
17	
18	const GaugeContainer = styled.div`

20	  border-radius: 12px;

22	  margin-bottom: 30px;
23	  box-shadow: 0 4px 15px rgba(0, 0, 0, 0.08);
24	`;
25	
26	const GaugeTitle = styled.h3`
27	  color: #2C3E50;
28	  margin-top: 0;
29	  font-size: 1.4em;
30	`;
31	
32	const GaugeValue = styled.div`
33	  font-size: 2.8em;
34	  font-weight: 700;
35	  color: ${props => props.color};
36	  margin: 10px 0;
37	`;
38	
39	const GaugeBar = styled.div`
40	  height: 30px;
41	  background-color: #e0e0e0;
42	  border-radius: 15px;
43	  overflow: hidden;
44	  margin-top: 15px;
45	`;
46	
47	const GaugeFill = styled.div<{ percentage: number; color: string }>`
48	  height: 100%;
49	  width: ${props => `${props.percentage}%`};
50	  background-color: ${props => props.color};
51	  transition: width 0.5s ease-in-out;
52	`;
53	
54	const StabilityGauge: React.FC<StabilityGaugeProps> = ({ data, title, unit, colorScheme, benchmark, maxValue }) => {
55	  let statusColor: 'success' | 'warning' | 'danger';
56	  let percentageValue: number;
57	
58	  if (title.includes('Latency')) {
59	    // Latency: Lower is better. Benchmark is 100ms.
60	    const latencyRatio = data.latencyMs / benchmark;
61	    if (data.latencyMs <= benchmark * 0.8) {
62	      statusColor = 'success'; // Very fast
63	    } else if (data.latencyMs <= benchmark * 1.2) {
64	      statusColor = 'warning'; // Near benchmark
65	    } else {
66	      statusColor = 'danger'; // Slow
67	    }
68	    percentageValue = Math.min(100, (data.latencyMs / benchmark) * 100);
69	
70	  } else if (title.includes('Visual Consistency')) {
71	    // VSC: Higher is better. Max is 100.
72	    percentageValue = Math.min(100, data.visualConsistencyScore);
73	    if (percentageValue >= 90) {
74	      statusColor = 'success';
75	    } else if (percentageValue >= 75) {
76	      statusColor = 'warning';
77	    } else {
78	      statusColor = 'danger';
79	    }
80	  }
81	
82	  return (
83	    <GaugeContainer>
84	      <GaugeTitle>{title}</GaugeTitle>
85	      <GaugeValue color={statusColor}>{percentageValue.toFixed(1)} {unit}</GaugeValue>
86	      <GaugeBar>
87	        <GaugeFill percentage={percentageValue} color={statusColor} />
88	      </GaugeBar>
89	      <p style={{ marginTop: '10px', fontSize: '0.9em', color: '#666' }}>
90	        Status: {statusColor.charAt(0).toUpperCase() + statusColor.slice(1)}
91	      </p>
92	    </GaugeContainer>
93	  );
94	};
95
